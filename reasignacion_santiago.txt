BLOQUE 2 — Ejecutar reasignacion (solo si Bloque 1 fue correcto)

\Illuminate\Support\Facades\DB::transaction(function() use ($asignacion, $equipoIds) {
    // Solo activamos la asignacion, NO tocamos cantidad (equipos ya existen en el sistema)
    $asignacion->update([
        'estatus' => \App\Models\Asignacion::EN_PROCESO,
    ]);
    $ok = 0;
    foreach ($equipoIds as $equipoId) {
        \App\Models\AsignacionEquipo::create([
            'asignacion_id' => $asignacion->id,
            'equipo_id'     => $equipoId,
            'camino'        => \App\Models\AsignacionEquipo::EN_PROCESO,
            'inicio_en'     => now(),
        ]);
        \App\Models\Equipo::where('id', $equipoId)->update([
            'estatus_ciclo' => \App\Models\Equipo::CICLO_PREPARACION,
            'estatus_area'  => \App\Models\Equipo::AREA_EN_PROCESO,
        ]);
        $ok++;
    }
    echo "OK: {$ok} equipos asignados a la asignacion #{$asignacion->id}\n";
});
